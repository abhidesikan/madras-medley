{{- /* Open Graph / Twitter Card meta. Image: featured_image param, else the first image in the content, else the site default. URLs are absolutised against the site BaseURL. No width/height is declared — let crawlers read the real image rather than guessing. */ -}} {{- $desc := .Site.Params.description -}} {{- with .Description }}{{ $desc = . }}{{ end -}} {{- $ogTitle := .Title -}} {{- if .IsHome }}{{ $ogTitle = .Site.Title }}{{ end -}} {{- $ogImage := "" -}} {{- if .Params.featured_image -}} {{- $ogImage = .Params.featured_image -}} {{- else if .IsPage -}} {{- $images := findRE "]+src=\"([^\"]+)\"" .Content 1 -}} {{- if $images -}} {{- $img := index $images 0 -}} {{- $ogImage = replaceRE ".*src=\"([^\"]+)\".*" "$1" $img -}} {{- end -}} {{- end -}} {{- if not $ogImage -}} {{- $ogImage = .Site.Params.defaultOGImage -}} {{- end -}} {{- if and $ogImage (not (hasPrefix $ogImage "http")) -}} {{- $ogImage = printf "%s%s" .Site.BaseURL (strings.TrimPrefix "/" $ogImage) -}} {{- end -}} {{- if $ogImage }} {{- end }} {{- if $ogImage }} {{- end }}
